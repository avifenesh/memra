pub fn validate_active_count(n_used: usize, active_count: usize) -> Result<(), String> {
    if active_count < n_used {
        return Err(format!(
            "sigmoid router requires active_count >= n_used: active_count={active_count}, n_used={n_used}",
        ));
    }
    Ok(())
}
