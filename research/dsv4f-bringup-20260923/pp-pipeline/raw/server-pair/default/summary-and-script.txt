#!/usr/bin/env bash
# q-v3e2.sh: the rest of q-v3e.sh (its row() read $1 after shift and died under set -u after
# the first row, plain-unset, which completed). Same binary (beae5264a build), after q-v3f.sh.
while ! grep -q V3F_DONE /root/rcpt/q-v3f.summary 2>/dev/null; do sleep 30; done
set -u
S=/root/rcpt/q-v3e.summary; R=/root/rcpt/pipe/default
B=/root/lane/target-pipe2/release/memra-server
wait_lock() { while ! flock -n /tmp/memra-gpu.lock true; do sleep 10; done; }
d=$R/plain-unset
echo "plain-unset rc=done-in-q-v3e lanes=[$(grep -h 'serving lane' $d/serve.log | cut -c1-160)] $(grep -hE 'CELL ' $d/controller.log 2>/dev/null | grep -v warmup | cut -c1-150 | tr '\n' ' ')" >> $S
row() {  # row <name> <cells> [env...]
  local name=$1 cells=$2; shift 2; local d=$R/$name; wait_lock
  /root/box/cell.sh $d $B $cells "$@" > $d.out 2>&1
  local rc=$?
  echo "$name rc=$rc lanes=[$(grep -h 'serving lane' $d/serve.log | cut -c1-160)] $(grep -hE 'CELL ' $d/controller.log 2>/dev/null | grep -v warmup | cut -c1-150 | tr '\n' ' ')" >> $S
}
row plain-one /root/box/cells-pipe2.txt MEMRA_DSV4_SESSIONS=1
row plain-unset-b /root/box/cells-pipe2.txt
row plain-one-b /root/box/cells-pipe2.txt MEMRA_DSV4_SESSIONS=1
row dspark-unset /root/box/cells-pipe.txt MEMRA_DSV4_DRAFTER=dspark
echo V3E_DONE >> $S
echo V3E2_DONE >> /root/rcpt/q-v3e2.summary
